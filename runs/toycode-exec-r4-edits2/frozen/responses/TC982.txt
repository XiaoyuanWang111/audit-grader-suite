def run_length(s):
    result = []
    prev = s[0]
    count = 1
    for ch in s[1:]:
        if ch == prev:
            count += 1
        else:
            result.append(prev + str(count))
            prev, count = ch, 1
    result.append(prev + str(count))
    return "".join(result)
